# Generate resources.rc: every file under assets/ becomes an RCDATA resource named by its relative
# path ("icons/24/play.svg"); app.ico is the window/exe icon (id 1).
set(ASSET_DIR ${CMAKE_SOURCE_DIR}/assets)
file(GLOB_RECURSE ST_ASSETS RELATIVE ${ASSET_DIR} CONFIGURE_DEPENDS ${ASSET_DIR}/*)
list(FILTER ST_ASSETS EXCLUDE REGEX "\.ico$")
# "_name" files are dev data, not app assets (i18n/_keys.json: tools/i18n_check.py --dump for translators).
list(FILTER ST_ASSETS EXCLUDE REGEX "(^|/)_[^/]*$")
set(RC_CONTENT "// Generated by src/app/CMakeLists.txt - do not edit\n1 ICON \"${ASSET_DIR}/app.ico\"\n")
set(ST_ASSET_FILES)
foreach(asset ${ST_ASSETS})
    string(APPEND RC_CONTENT "\"${asset}\" RCDATA \"${ASSET_DIR}/${asset}\"\n")
    list(APPEND ST_ASSET_FILES ${ASSET_DIR}/${asset})
endforeach()
# Version resource: Windows shows FileDescription as the app name (media flyout / lock screen / Task Manager).
string(APPEND RC_CONTENT
    "1 VERSIONINFO\n"
    "FILEVERSION ${PROJECT_VERSION_MAJOR},${PROJECT_VERSION_MINOR},${PROJECT_VERSION_PATCH},0\n"
    "PRODUCTVERSION ${PROJECT_VERSION_MAJOR},${PROJECT_VERSION_MINOR},${PROJECT_VERSION_PATCH},0\n"
    "FILEOS 0x40004\nFILETYPE 0x1\nBEGIN\n"
    "  BLOCK \"StringFileInfo\"\n  BEGIN\n    BLOCK \"040904B0\"\n    BEGIN\n"
    "      VALUE \"CompanyName\", \"ShadeTube\"\n"
    "      VALUE \"FileDescription\", \"ShadeTube\"\n"
    "      VALUE \"ProductName\", \"ShadeTube\"\n"
    "      VALUE \"InternalName\", \"ShadeTube\"\n"
    "      VALUE \"OriginalFilename\", \"ShadeTube.exe\"\n"
    "      VALUE \"FileVersion\", \"${PROJECT_VERSION}\"\n"
    "      VALUE \"ProductVersion\", \"${PROJECT_VERSION}\"\n"
    "    END\n  END\n"
    "  BLOCK \"VarFileInfo\"\n  BEGIN\n    VALUE \"Translation\", 0x409, 1200\n  END\nEND\n")
file(GENERATE OUTPUT ${CMAKE_CURRENT_BINARY_DIR}/resources.rc CONTENT "${RC_CONTENT}")
set_source_files_properties(${CMAKE_CURRENT_BINARY_DIR}/resources.rc PROPERTIES OBJECT_DEPENDS "${ST_ASSET_FILES}")

# windowsapp: C++/WinRT SystemMediaTransportControls (Smtc). bcrypt: MD5 for Last.fm signatures. crypt32: DPAPI.
target_link_libraries(ShadeTube PRIVATE st_ui st_player st_lyrics st_musicbrainz st_spotify st_youtube st_catalog st_audio st_core webview2
    windowsapp bcrypt crypt32)
# "ShadeTube 0.3" in Ayarlar > Hakkında.
target_compile_definitions(ShadeTube PRIVATE ST_VERSION_SHORT="${PROJECT_VERSION_MAJOR}.${PROJECT_VERSION_MINOR}")
# The WebView2 static loader is built with the release static CRT; in a Debug (/MTd) build let the debug CRT win.
target_link_options(ShadeTube PRIVATE $<$<CONFIG:Debug>:/NODEFAULTLIB:libcmt>)
set_target_properties(ShadeTube PROPERTIES RUNTIME_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/bin)
