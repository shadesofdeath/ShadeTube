# ShadeTubeSetup.exe: the per-user setup with its own window on st_ui, carrying the app as a compressed, sealed payload
# (docs/DEVELOPMENT.md "Setup").
#   st_setup_core   payload format, command line, the install / uninstall work (shared with setup_test)
#   setup_pack      build-host tool: packs the app's build output into payload.bin, trims the translations
#   ShadeTubeSetup  the setup itself (bin\ShadeTubeSetup.exe; tools/package.ps1 ships it as ShadeTube-<ver>-setup.exe)
add_library(st_setup_core STATIC
    Payload.cpp Payload.h
    SetupArgs.cpp SetupArgs.h
    SetupEngine.cpp SetupEngine.h
)
target_link_libraries(st_setup_core PUBLIC st_install PRIVATE cabinet bcrypt)

add_executable(setup_pack Pack.cpp)
target_link_libraries(setup_pack PRIVATE st_setup_core version)

# ---- The payload: ShadeTube.exe of this build tree and the licenses that travel with it ----------------------------
# LZMS (Windows' best ratio, fast to unpack) for release builds; XPRESS_HUFF keeps Debug builds quick to pack.
set(_payload ${CMAKE_CURRENT_BINARY_DIR}/payload.bin)
add_custom_command(OUTPUT ${_payload}
    COMMAND setup_pack payload ${_payload} ${PROJECT_VERSION} $<IF:$<CONFIG:Debug>,xpress-huff,lzms>
            ShadeTube.exe=$<TARGET_FILE:ShadeTube>
            LICENSE.txt=${CMAKE_SOURCE_DIR}/LICENSE
            THIRD_PARTY_NOTICES.md=${CMAKE_SOURCE_DIR}/THIRD_PARTY_NOTICES.md
            OFL.txt=${CMAKE_SOURCE_DIR}/assets/fonts/OFL.txt
    DEPENDS setup_pack ShadeTube ${CMAKE_SOURCE_DIR}/LICENSE ${CMAKE_SOURCE_DIR}/THIRD_PARTY_NOTICES.md
            ${CMAKE_SOURCE_DIR}/assets/fonts/OFL.txt
    COMMENT "Packing ShadeTube.exe into the setup's payload"
    VERBATIM)

# ---- Translations: only the texts the setup uses (its sources + what it links that calls tr()) ----------------------
file(GLOB _setup_tr_sources CONFIGURE_DEPENDS ${CMAKE_CURRENT_SOURCE_DIR}/*.cpp ${CMAKE_SOURCE_DIR}/src/ui/*.cpp
     ${CMAKE_SOURCE_DIR}/src/gfx/*.cpp ${CMAKE_SOURCE_DIR}/src/core/*.cpp)
list(REMOVE_ITEM _setup_tr_sources ${CMAKE_CURRENT_SOURCE_DIR}/Pack.cpp)
list(APPEND _setup_tr_sources ${CMAKE_SOURCE_DIR}/src/app/Installer.cpp ${CMAKE_SOURCE_DIR}/src/app/Updater.cpp
     ${CMAKE_SOURCE_DIR}/src/app/Autostart.cpp)
file(GLOB _i18n_files CONFIGURE_DEPENDS ${CMAKE_SOURCE_DIR}/assets/i18n/*.json)
list(FILTER _i18n_files EXCLUDE REGEX "/_[^/]*$")
set(_i18n_out ${CMAKE_CURRENT_BINARY_DIR}/i18n)
set(_i18n_outputs)
foreach(_f ${_i18n_files})
    get_filename_component(_n ${_f} NAME)
    list(APPEND _i18n_outputs ${_i18n_out}/${_n})
endforeach()
add_custom_command(OUTPUT ${_i18n_outputs}
    COMMAND setup_pack i18n ${_i18n_out} ${CMAKE_SOURCE_DIR}/assets/i18n ${_setup_tr_sources}
    DEPENDS setup_pack ${_i18n_files} ${_setup_tr_sources}
    COMMENT "Trimming the translations for the setup"
    VERBATIM)

# ---- Resources: the app's icon, fonts, tokens, the icons it draws, the logo, translations, payload, version ------------
set(_assets ${CMAKE_SOURCE_DIR}/assets)
set(_rc "// Generated by src/setup/CMakeLists.txt - do not edit\n1 ICON \"${_assets}/app.ico\"\n")
set(_rc_deps ${_assets}/app.ico)
foreach(_file fonts/BricolageGrotesque.ttf fonts/JetBrainsMono.ttf tokens/tokens.dark.json tokens/tokens.light.json
              logo/logo-mark.svg)
    string(APPEND _rc "\"${_file}\" RCDATA \"${_assets}/${_file}\"\n")
    list(APPEND _rc_deps ${_assets}/${_file})
endforeach()
foreach(_grid 12 14 16 18 20 24)
    foreach(_icon minimize close folder download refresh trash check warning error info play)
        string(APPEND _rc "\"icons/${_grid}/${_icon}.svg\" RCDATA \"${_assets}/icons/${_grid}/${_icon}.svg\"\n")
        list(APPEND _rc_deps ${_assets}/icons/${_grid}/${_icon}.svg)
    endforeach()
endforeach()
foreach(_out ${_i18n_outputs})
    get_filename_component(_n ${_out} NAME)
    string(APPEND _rc "\"i18n/${_n}\" RCDATA \"${_out}\"\n")
endforeach()
string(APPEND _rc "\"payload\" RCDATA \"${_payload}\"\n")
# ProductName is "ShadeTube Setup", never "ShadeTube": the updater's verifyExe() must never take the setup for the app.
string(APPEND _rc
    "1 VERSIONINFO\n"
    "FILEVERSION ${PROJECT_VERSION_MAJOR},${PROJECT_VERSION_MINOR},${PROJECT_VERSION_PATCH},0\n"
    "PRODUCTVERSION ${PROJECT_VERSION_MAJOR},${PROJECT_VERSION_MINOR},${PROJECT_VERSION_PATCH},0\n"
    "FILEOS 0x40004\nFILETYPE 0x1\nBEGIN\n"
    "  BLOCK \"StringFileInfo\"\n  BEGIN\n    BLOCK \"040904B0\"\n    BEGIN\n"
    "      VALUE \"CompanyName\", \"ShadeTube\"\n"
    "      VALUE \"FileDescription\", \"ShadeTube Setup\"\n"
    "      VALUE \"ProductName\", \"ShadeTube Setup\"\n"
    "      VALUE \"InternalName\", \"ShadeTubeSetup\"\n"
    "      VALUE \"OriginalFilename\", \"ShadeTubeSetup.exe\"\n"
    "      VALUE \"FileVersion\", \"${PROJECT_VERSION}\"\n"
    "      VALUE \"ProductVersion\", \"${PROJECT_VERSION}\"\n"
    "    END\n  END\n"
    "  BLOCK \"VarFileInfo\"\n  BEGIN\n    VALUE \"Translation\", 0x409, 1200\n  END\nEND\n")
file(GENERATE OUTPUT ${CMAKE_CURRENT_BINARY_DIR}/setup.rc CONTENT "${_rc}")
set_source_files_properties(${CMAKE_CURRENT_BINARY_DIR}/setup.rc PROPERTIES OBJECT_DEPENDS "${_rc_deps};${_payload};${_i18n_outputs}")

add_executable(ShadeTubeSetup WIN32
    main.cpp
    SetupUi.cpp SetupUi.h
    ShadeTubeSetup.manifest
    ${CMAKE_CURRENT_BINARY_DIR}/setup.rc
    ${_payload} ${_i18n_outputs}   # generated before the resources are compiled
)
set_source_files_properties(${_payload} ${_i18n_outputs} PROPERTIES HEADER_FILE_ONLY TRUE)
target_link_libraries(ShadeTubeSetup PRIVATE st_setup_core st_ui st_install st_core)
set_target_properties(ShadeTubeSetup PROPERTIES RUNTIME_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/bin)
