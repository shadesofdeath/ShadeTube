# Windows shell integration test. The ShadeTube exe cannot be linked, so app/WinShell.cpp (+ app/Installer.cpp and the
# app/Updater.cpp it uses) are compiled in directly. The resources WinShell reads are embedded like in the app: app.ico
# as icon 1 and the SVG glyphs of the taskbar buttons and jump-list tasks as RCDATA named by their asset path.
set(_assets ${CMAKE_SOURCE_DIR}/assets)
set(_rc "// Generated by tests/winshell/CMakeLists.txt - do not edit\n1 ICON \"${_assets}/app.ico\"\n")
set(_rc_deps ${_assets}/app.ico)
foreach(_grid 16 20 24)
    foreach(_icon play pause prev next mini-player)
        string(APPEND _rc "\"icons/${_grid}/${_icon}.svg\" RCDATA \"${_assets}/icons/${_grid}/${_icon}.svg\"\n")
        list(APPEND _rc_deps ${_assets}/icons/${_grid}/${_icon}.svg)
    endforeach()
endforeach()
file(GENERATE OUTPUT ${CMAKE_CURRENT_BINARY_DIR}/winshell_test.rc CONTENT "${_rc}")
set_source_files_properties(${CMAKE_CURRENT_BINARY_DIR}/winshell_test.rc PROPERTIES OBJECT_DEPENDS "${_rc_deps}")

add_executable(winshell_test winshell_test.cpp
    ${CMAKE_SOURCE_DIR}/src/app/WinShell.cpp
    ${CMAKE_SOURCE_DIR}/src/app/Installer.cpp
    ${CMAKE_SOURCE_DIR}/src/app/Updater.cpp
    ${CMAKE_CURRENT_BINARY_DIR}/winshell_test.rc)
target_link_libraries(winshell_test PRIVATE st_core)
