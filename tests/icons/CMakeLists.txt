# The icon set (assets/icons, written by tools/icons.py) through gfx::Icons: every file parses and follows the rules
# (viewBox = its grid, one stroke weight, black only), the grid choice and the old "-thin" names, crisp 1 px strokes
# at the native sizes, the 1 px floor below them, filled glyphs covering their outlines; `--sheet <dir>` renders
# contact sheets (every icon at 12-24 DIP, 80-175 %, dark / light / preset themes, magnified pixels) to look at.
# Links st_gfx; the icons, fonts, tokens and themes are embedded like in the app (RCDATA named by asset path).
set(_assets ${CMAKE_SOURCE_DIR}/assets)
file(GLOB_RECURSE _icon_files RELATIVE ${_assets} CONFIGURE_DEPENDS ${_assets}/icons/*.svg)
set(_rc "// Generated by tests/icons/CMakeLists.txt - do not edit\n")
set(_rc_deps "")
foreach(_file fonts/BricolageGrotesque.ttf fonts/JetBrainsMono.ttf tokens/tokens.dark.json tokens/tokens.light.json
              ${_icon_files})
    string(APPEND _rc "\"${_file}\" RCDATA \"${_assets}/${_file}\"\n")
    list(APPEND _rc_deps ${_assets}/${_file})
endforeach()
file(GENERATE OUTPUT ${CMAKE_CURRENT_BINARY_DIR}/icons_test.rc CONTENT "${_rc}")
set_source_files_properties(${CMAKE_CURRENT_BINARY_DIR}/icons_test.rc PROPERTIES OBJECT_DEPENDS "${_rc_deps}")

add_executable(icons_test icons_test.cpp ${CMAKE_CURRENT_BINARY_DIR}/icons_test.rc)
target_link_libraries(icons_test PRIVATE st_gfx)
target_compile_definitions(icons_test PRIVATE ST_ASSET_DIR="${_assets}")
