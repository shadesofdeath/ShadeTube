# The overlay family (ui/Popups, ui/Acrylic, the Window's overlay layer) on a hidden window. Links st_ui; the assets it
# draws with (fonts, tokens, the noise tile, a few icons) are embedded like in the app (RCDATA named by asset path).
set(_assets ${CMAKE_SOURCE_DIR}/assets)
set(_rc "// Generated by tests/popups/CMakeLists.txt - do not edit\n")
set(_rc_deps "")
foreach(_file fonts/BricolageGrotesque.ttf fonts/JetBrainsMono.ttf tokens/tokens.dark.json tokens/tokens.light.json
              textures/noise-tile.png)
    string(APPEND _rc "\"${_file}\" RCDATA \"${_assets}/${_file}\"\n")
    list(APPEND _rc_deps ${_assets}/${_file})
endforeach()
foreach(_grid 12 14 16 18 20 24)
    foreach(_icon check chevron-right plus info error warning queue playlist)
        string(APPEND _rc "\"icons/${_grid}/${_icon}.svg\" RCDATA \"${_assets}/icons/${_grid}/${_icon}.svg\"\n")
        list(APPEND _rc_deps ${_assets}/icons/${_grid}/${_icon}.svg)
    endforeach()
endforeach()
file(GENERATE OUTPUT ${CMAKE_CURRENT_BINARY_DIR}/popups_test.rc CONTENT "${_rc}")
set_source_files_properties(${CMAKE_CURRENT_BINARY_DIR}/popups_test.rc PROPERTIES OBJECT_DEPENDS "${_rc_deps}")

add_executable(popups_test popups_test.cpp ${CMAKE_CURRENT_BINARY_DIR}/popups_test.rc)
target_link_libraries(popups_test PRIVATE st_ui)
